--- Makefile ---
SHELL := /bin/bash
VENV := .venv
PYTHON := $(VENV)/bin/python
PIP := $(VENV)/bin/pip
BENCH := $(VENV)/bin/benchmark

DATA_ROOT ?= data/raw/fashion-product-images-small
SPLIT_FILE ?= $(DATA_ROOT)/splits/test.json
CSV ?= data/raw/fashion-product-images-small/styles.csv
IMAGES_ROOT ?= data/raw/fashion-product-images-small/images
DATASET_NAME ?= fashion-product-small
DEVICE ?= cpu
MODELS ?= all
BATCH_SIZE ?= 64
OUTPUT ?= outputs

LABEL_FIELD ?= subCategory
TRAIN_RATIO ?= 0.8
SEED ?= 42
NO_CACHE ?= false
NO_LATENCY ?= false

NO_CACHE_FLAG := $(if $(filter true,$(NO_CACHE)),--no-cache,)
NO_LATENCY_FLAG := $(if $(filter true,$(NO_LATENCY)),--no-latency,)

.PHONY: all help venv install prepare-fashion-product benchmark clean

all: help

help:
	@printf "Available targets:\n"
	@printf "  make venv                  Create the virtual environment and install package dependencies.\n"
	@printf "  make install               Alias for make venv.\n"
	@printf "  make prepare-fashion-product\n"
	@printf "                            Convert the Kaggle Fashion Product Images Small dataset into the benchmark split format.\n"
	@printf "  make benchmark             Run the benchmark on the prepared dataset.\n"
	@printf "  make clean                 Remove the local virtual environment and build artifacts.\n"
	@printf "\n"
	@printf "Optional variables: DATA_ROOT, SPLIT_FILE, CSV, IMAGES_ROOT, DATASET_NAME, MODELS, DEVICE, BATCH_SIZE, OUTPUT, NO_CACHE, NO_LATENCY\n"

venv:
	test -d $(VENV) || python3 -m venv $(VENV)
	$(PYTHON) -m pip install --upgrade pip setuptools wheel
	$(PIP) install -e .

install: venv

prepare-fashion-product: venv
	$(PYTHON) scripts/prepare_fashion_product.py \
		--csv $(CSV) \
		--images-root $(IMAGES_ROOT) \
		--output-root $(DATA_ROOT) \
		--label-field $(LABEL_FIELD) \
		--train-ratio $(TRAIN_RATIO) \
		--seed $(SEED)

benchmark: venv
	$(BENCH) benchmark \
		--dataset-root $(DATA_ROOT) \
		--split-file $(SPLIT_FILE) \
		--name $(DATASET_NAME) \
		--device $(DEVICE) \
		--models $(MODELS) \
		--batch-size $(BATCH_SIZE) \
		$(NO_CACHE_FLAG) \
		$(NO_LATENCY_FLAG) \
		--output $(OUTPUT)

clean:
	rm -rf $(VENV) outputs
